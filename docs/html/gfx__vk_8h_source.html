89static inline auto FindQueueFamilies(const VkPhysicalDevice& device, const VkSurfaceKHR& surface) -> QueueFamilyIndices {
90 QueueFamilyIndices indices{};
91 uint32_t count = 0;
92 vkGetPhysicalDeviceQueueFamilyProperties(device, &count, nullptr);
93 std::vector<VkQueueFamilyProperties> families(count);
94 vkGetPhysicalDeviceQueueFamilyProperties(device, &count, &families[0]);
95 auto idx = 0;
96 for (const auto& family : families) {
97 if (family.queueFlags & VK_QUEUE_GRAPHICS_BIT) {
98 indices.graphics = idx;
99 }
100
101 VkBool32 present_supported = false;
102 vkGetPhysicalDeviceSurfaceSupportKHR(device, idx, surface, &present_supported);
103 if (present_supported) {
104 indices.present = idx;
105 }
106
107 if (indices.IsComplete()) {
108 break;
109 }
110
111 idx++;
112 }
113 return indices;
114}

116static inline auto FindSuitablePhysicalDevice(const VkInstance& instance, VkPhysicalDevice* result,
117 const std::function<bool(const VkPhysicalDevice&)>& predicate) -> bool {
118 uint32_t count = 0;
119 vkEnumeratePhysicalDevices(instance, &count, nullptr);
120 LOG_IF(FATAL, count == 0) << "failed to find GPUs w/ vulkan support.";
121 std::vector<VkPhysicalDevice> devices(count);
122 vkEnumeratePhysicalDevices(instance, &count, &devices[0]);
123 const auto device = std::ranges::find_if(devices, predicate);
124 if (device == std::end(devices)) {
125 (*result) = VK_NULL_HANDLE;
126 return false;
127 }
128 (*result) = (*device);
129 ASSERT((*result) != VK_NULL_HANDLE);
130 return true;
131}

133static inline auto FindSupportedFormat(const VkPhysicalDevice physical_device, const std::vector<VkFormat>& formats,
134 VkImageTiling tiling, VkFormatFeatureFlags features) -> VkFormat {
135 for (const auto& format : formats) {
136 VkFormatProperties properties{};
137 vkGetPhysicalDeviceFormatProperties(physical_device, format, &properties);
138 if (tiling == VK_IMAGE_TILING_LINEAR && (properties.linearTilingFeatures & features) == features) {
139 return format;
140 } else if (tiling == VK_IMAGE_TILING_OPTIMAL && (properties.optimalTilingFeatures & features) == features) {
141 return format;
142 }
143 }
144 LOG(FATAL) << "failed to find supported depth format";
145}

169static inline auto FindMemoryType(const VkPhysicalDevice& physical_device, const uint32_t filter,
170 const VkMemoryPropertyFlags& flags) -> uint32_t {
171 VkPhysicalDeviceMemoryProperties mem_properties{};
172 vkGetPhysicalDeviceMemoryProperties(physical_device, &mem_properties);
173 for (auto idx = 0; idx < mem_properties.memoryTypeCount; idx++) {
174 if ((filter & (1 << idx)) && ((mem_properties.memoryTypes[idx].propertyFlags & flags) == flags))
175 return idx;
176 }
177 LOG(FATAL) << "failed to find suitable memory for physical device.";
178}

180static inline void CopyVkBuffer(const VkDevice& device, const VkCommandPool& command_pool, const VkQueue& queue, VkBuffer& dst,
181 const VkBuffer& src, const VkDeviceSize size) {
182 VkCommandBufferAllocateInfo alloc_info{};
183 alloc_info.sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_ALLOCATE_INFO;
184 alloc_info.level = VK_COMMAND_BUFFER_LEVEL_PRIMARY;
185 alloc_info.commandPool = command_pool;
186 alloc_info.commandBufferCount = 1;
187
188 VkCommandBuffer command_buffer{};
189 CHECK_VK(FATAL, vkAllocateCommandBuffers(device, &alloc_info, &command_buffer), "failed to allocate vk command buffer");
190
191 VkCommandBufferBeginInfo begin_info{};
192 begin_info.sType = VK_STRUCTURE_TYPE_COMMAND_BUFFER_BEGIN_INFO;
193 begin_info.flags = VK_COMMAND_BUFFER_USAGE_ONE_TIME_SUBMIT_BIT;
194 vkBeginCommandBuffer(command_buffer, &begin_info);
195 {
196 VkBufferCopy copy_region{};
197 copy_region.srcOffset = 0;
198 copy_region.dstOffset = 0;
199 copy_region.size = size;
200 vkCmdCopyBuffer(command_buffer, src, dst, 1, &copy_region);
201 }
202 vkEndCommandBuffer(command_buffer);
203
204 VkSubmitInfo submit_info{};
205 submit_info.sType = VK_STRUCTURE_TYPE_SUBMIT_INFO;
206 submit_info.commandBufferCount = 1;
207 submit_info.pCommandBuffers = &command_buffer;
208
209 CHECK_VK(FATAL, vkQueueSubmit(queue, 1, &submit_info, VK_NULL_HANDLE), "failed to submit to graphics queue");
210 CHECK_VK(FATAL, vkQueueWaitIdle(queue), "failed to wait for graphics queue idle");
211 vkFreeCommandBuffers(device, command_pool, 1, &command_buffer);
212}
