185 const auto idx = rpos_ + offset;

193 inline auto NextChar() -> char {
194 if ((rpos_ + 1) > wpos_) {

196 return EOF;

199 const auto c = buffer_.at(rpos_);
200 rpos_ += 1;
201 switch (c) {
202 case '\n':
203 pos_.row += 1;
204 pos_.column = 1;
205 return c;
206 default:
207 pos_.column += 1;
208 return c;
209 }

262 inline auto ParseNumber(T* result) -> bool {
263 token_len_ = 0;

268 } else if (next == 'x' || next == 'X') {
269 break;

276 } while (true);

279 return false;

282 return true;
283 }
