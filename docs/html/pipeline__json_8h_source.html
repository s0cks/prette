163 auto String(const char* str, SizeType length, bool copy) -> bool {
164 const auto value = std::string(str, length);
165 switch (GetState()) {
166 case kParsingType: {
167 if (value != "Pipeline")
168 return Error(fmt::format("unexpected value for 'type' field: `{}`", value));
169 return TransitionTo(kOpenDoc);
170 }
171 case kParsingName:
172 DLOG(INFO) << "parsed name: " << value;
173 return TransitionTo(kParsingMeta);
174 case kParsingCullMode:
175 OnParseRasterizerCullMode(value);

179 return TransitionTo(kParsingData);

209 auto Key(const char* str, SizeType length, bool copy) -> bool {
210 const auto key = std::string(str, length);
211 switch (GetState()) {
212 case kOpenDoc: {
213 static const std::array<KeyState, 3> kPipelineKeys = {
214 KeyState{.name = "type", .state = kParsingType},
215 KeyState{.name = "meta", .state = kExpectMeta},
216 KeyState{.name = "data", .state = kExpectData},
217 };
218 return CheckKeyTable(key, kPipelineKeys);
219 }
220 case kParsingMeta: {
221 static const std::array<KeyState, 1> kMetaKeys = {
222 KeyState{.name = "name", .state = kParsingName},
223 };
224 return CheckKeyTable(key, kMetaKeys);
225 }
226 case kParsingData: {
227 static const std::array<KeyState, 5> kDataKeys = {
228 KeyState{.name = "cullMode", .state = kParsingCullMode},
229 KeyState{.name = "frontFace", .state = kParsingFrontFace},
230 KeyState{.name = "vertexShader", .state = kParsingVertexShader},
231 KeyState{.name = "fragmentShader", .state = kParsingFragmentShader},
232 KeyState{.name = "dynamicStates", .state = kExpectDynamicStates},
233 };
234 return CheckKeyTable(key, kDataKeys);
235 }
236 default:
237 break;
238 }
239 return InvalidState();
240 }

281inline auto ReadJsonFromFile(fs::path path, std::string& result) -> bool {
282 if (!fs::exists(path)) {
283 LOG(ERROR) << "failed to find ShaderCode at: " << path;
284 return false;
285 }
286 std::ifstream file(path, std::ios::ate | std::ios::binary);
287 if (!file.is_open()) {
288 LOG(ERROR) << "failed to open ShaderCode at: " << path;
289 return false;
290 }
291 auto filesize = static_cast<std::streamsize>(file.tellg());
292 result.resize(filesize + 1);
293 file.seekg(0);
294 file.read((char*)result.data(), filesize);
295 file.close();
296 result[filesize] = '\0';
297 return true;
298}

303 LOG_IF(FATAL, !ReadJsonFromFile(path, buffer)) << "failed to read json from file: " << path;
304 json::Reader reader{};
305 json::StringStream ss(buffer.data());
306 if (!reader.Parse(ss, handler)) {
307 json::ParseErrorCode e = reader.GetParseErrorCode();
308 size_t o = reader.GetErrorOffset();
309 std::cerr << "Error: ";
310 if (handler.HasError()) {
311 std::cerr << handler.GetError();
312 } else {
313 std::cerr << json::GetParseError_En(e);
314 }
315 std::cerr << std::endl;
316 std::cerr << " at offset " << o << " near '" << std::string(buffer.data()).substr(o, 10) << "...'" << std::endl;
317 }
